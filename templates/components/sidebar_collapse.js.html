(function () { var nav = document.getElementById('sidebarNav'); if (!nav) return; var toggles = Array.prototype.slice.call(nav.querySelectorAll('.sidebar-menu-toggle')); if (toggles.length === 0) return; var role = nav.getAttribute('data-role') || 'app'; var KEY = 'eduPro_sidebar_' + role + '_open'; function submenu(btn) { var id = btn.getAttribute('aria-controls'); return id ? document.getElementById(id) : null; } function toggleFor(sub) { for (var i = 0; i < toggles.length; i++) { if (toggles[i].getAttribute('aria-controls') === sub.id) return toggles[i]; } return null; } function sameGroup(a, b) { return a.parentNode === b.parentNode; } // Returns true when the state actually changed, so callers can tell a real // transition from a no-op and avoid redundant work. function setState(btn, open) { if (btn.classList.contains('open') === open) return false; btn.classList.toggle('open', open); btn.setAttribute('aria-expanded', open ? 'true' : 'false'); var sub = submenu(btn); if (sub) sub.classList.toggle('open', open); return true; } function persist() { var map = {}; toggles.forEach(function (t) { map[t.dataset.menu] = t.classList.contains('open'); }); try { localStorage.setItem(KEY, JSON.stringify(map)); } catch (e) {} } // Open a toggle while closing only its siblings — ancestors and unrelated // branches stay untouched, so nested submenus stay intact. function openExclusive(btn) { toggles.forEach(function (other) { if (other !== btn && sameGroup(other, btn)) setState(other, false); }); setState(btn, true); } // Walk from the outermost group down to the one holding `link` and open each. function expandAncestors(link) { var chain = []; var sub = link && link.closest ? link.closest('.sidebar-submenu') : null; while (sub) { chain.unshift(sub); var parent = sub.parentNode; sub = parent && parent.closest ? parent.closest('.sidebar-submenu') : null; } chain.forEach(function (s) { var btn = toggleFor(s); if (btn) openExclusive(btn); }); } function setActiveLink(link) { nav.querySelectorAll('a.nav-item.active').forEach(function (el) { el.classList.remove('active'); }); if (link) link.classList.add('active'); } // Pick the single best link for the current URL: the LONGEST href that // matches. Without the length check a parent link such as /academics/ // prefix-matches every page in the module and steals .active from the real // current page, depending only on DOM order. function bestMatch(path) { var best = null; var bestLen = -1; nav.querySelectorAll('a.nav-item[href]').forEach(function (link) { var href = link.getAttribute('href'); if (!href || href === '/') return; var endsClean = href.charAt(href.length - 1) === '/'; var hit = (path === href) || (endsClean && path.indexOf(href) === 0); if (hit && href.length > bestLen) { best = link; bestLen = href.length; } }); return best; } toggles.forEach(function (btn) { btn.addEventListener('click', function (e) { // A group header only ever expands or collapses — it must never navigate. e.preventDefault(); if (btn.classList.contains('open')) { setState(btn, false); } else { openExclusive(btn); } persist(); }); }); // 1. Restore the layout the user last chose, on every page. var saved = {}; try { saved = JSON.parse(localStorage.getItem(KEY)) || {}; } catch (e) { saved = {}; } // Applied in DOM order so parents are always restored before their children. toggles.forEach(function (t) { if (saved[t.dataset.menu] === true) setState(t, true); }); // 2. Then reveal where we are: highlight the current link and open the // groups that contain it. var current = bestMatch(window.location.pathname); if (current) { setActiveLink(current); expandAncestors(current); } // Nothing above persists. Only a real click saves state, so simply visiting a // page can no longer overwrite the open/closed layout the user picked. // Expose for AJAX navigation script window.sidebarSetActive = setActiveLink; })();
